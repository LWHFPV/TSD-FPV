/* DJI RC 2 bulk interface and DUML messages are documented by
 * https://github.com/v1kr4m-ai/DJI_RC2_bridge (MIT, see notice below).
 */
const rc2Usb = (() => {
  const filter = {vendorId: 0x2ca3, productId: 0x1021};
  const id = 'DJI RC 2 · USB';
  const index = -1;
  const enableSimulator = Uint8Array.of(0x55, 0x0e, 0x04, 0x66, 0x0a, 0x06, 0xeb, 0x34, 0x40, 0x06, 0x24, 0x01, 0xd9, 0xec);
  const readSticks = Uint8Array.of(0x55, 0x0d, 0x04, 0x33, 0x0a, 0x06, 0xeb, 0x34, 0x40, 0x06, 0x01, 0x74, 0x24);
  let active;
  let button;
  let output;
  let onChange;
  let rememberedDevice;
  let permissionRevision = 0;

  function forgetRememberedDevice() {
    rememberedDevice = undefined;
    permissionRevision++;
  }

  function refreshPermission() {
    const revision = ++permissionRevision;
    navigator.usb.getDevices().then(devices => {
      if (revision === permissionRevision) {
        rememberedDevice = devices.find(device => device.vendorId === filter.vendorId && device.productId === filter.productId);
      }
    }).catch(() => {});
  }

  function crc(bytes, end, value, polynomial) {
    for (let i = 0; i < end; i++) {
      value ^= bytes[i];
      for (let bit = 0; bit < 8; bit++) value = (value >>> 1) ^ ((value & 1) ? polynomial : 0);
    }
    return value;
  }

  function readAxes(session, chunk) {
    const bytes = new Uint8Array(session.tail.length + chunk.byteLength);
    bytes.set(session.tail);
    bytes.set(new Uint8Array(chunk.buffer, chunk.byteOffset, chunk.byteLength), session.tail.length);
    let offset = 0;
    while (offset < bytes.length) {
      if (bytes[offset] !== 0x55) { offset++; continue; }
      if (bytes.length - offset < 4) break;
      const header = bytes.subarray(offset, offset + 4);
      const length = header[1] | ((header[2] & 3) << 8);
      if (length < 13 || (header[2] >> 2) !== 1 || crc(header, 3, 0x77, 0x8c) !== header[3]) {
        offset++;
        continue;
      }
      if (bytes.length - offset < length) break;
      const packet = bytes.subarray(offset, offset + length);
      if (crc(packet, length - 2, 0x3692, 0x8408) !== (packet[length - 2] | packet[length - 1] << 8)) {
        offset++;
        continue;
      }
      offset += length;
      if (length !== 32 || packet[9] !== 6 || packet[10] !== 1) continue;
      // Report up as negative on vertical stick axes, as in the browser Gamepad API.
      const raw = [21, 18, 12, 15, 24].map(i => packet[i] | packet[i + 1] << 8);
      if (raw.some(value => value < 200 || value > 1900)) continue;
      session.axes = raw.map((value, axis) => {
        const normalized = Math.max(-1, Math.min(1, (value - 1024) / 660));
        return axis === 1 || axis === 3 ? -normalized : normalized;
      });
      // The reference format reserves a sixth 16-bit channel. Promote it only
      // after observing both a centered value and movement in the valid range.
      const zoomRaw = packet[27] | packet[28] << 8;
      const zoomValid = zoomRaw >= 200 && zoomRaw <= 1900;
      const zoom = zoomValid ? Math.max(-1, Math.min(1, (zoomRaw - 1024) / 660)) : 0;
      if (zoomValid && Math.abs(zoom) <= .08) session.zoomCentered = true;
      const detectedZoom = !session.zoomDetected && session.zoomCentered && Math.abs(zoom) >= .18;
      if (detectedZoom) session.zoomDetected = true;
      if (session.zoomDetected) session.axes.push(zoom);
      const first = !session.lastInput;
      session.lastInput = performance.now();
      if (first || detectedZoom) {
        output.value = session.zoomDetected ? '스틱 4축·짐벌·줌 다이얼 입력 수신 중' : '스틱 4축·짐벌 다이얼 입력 수신 중';
        onChange(true);
      }
    }
    session.tail = bytes.slice(offset);
  }

  function finish(session, message) {
    if (active !== session) return;
    active = undefined;
    clearInterval(session.watchdog);
    button.disabled = true;
    output.value = message;
    onChange(false);
    Promise.resolve(session.device?.opened ? session.device.close() : undefined).catch(() => {}).finally(() => {
      if (active) return;
      button.disabled = false;
      button.textContent = 'RC 2 USB 연결';
    });
  }

  async function connect() {
    if (active) {
      finish(active, 'RC 2 USB 연결 해제');
      return;
    }
    const session = {tail: new Uint8Array(0), lastInput: 0};
    active = session;
    button.disabled = true;
    let stage = 'select';
    output.value = '목록의 장치 이름 줄을 클릭한 뒤 아래의 연결 버튼을 누르세요. RC 2가 모델 코드로 표시될 수 있습니다.';
    try {
      const device = rememberedDevice || await navigator.usb.requestDevice({filters: [filter]});
      if (active !== session) return;
      permissionRevision++;
      rememberedDevice = device;
      session.device = device;
      stage = 'open';
      output.value = 'USB 장치 선택 완료 · 입력 인터페이스 여는 중';
      await device.open();
      if (active !== session) { await device.close(); return; }
      stage = 'configuration';
      const config = device.configuration || device.configurations[0];
      if (!config) throw new Error('configuration');
      if (!device.configuration) await device.selectConfiguration(config.configurationValue);
      stage = 'interface';
      const iface = config.interfaces.find(item => item.interfaceNumber === 0);
      const alt = iface?.alternates.find(item => item.interfaceClass === 0xff && item.interfaceSubclass === 0x43)
        || iface?.alternates.find(item => item.interfaceClass === 0xff);
      const input = alt?.endpoints.find(ep => ep.type === 'bulk' && ep.direction === 'in');
      const send = alt?.endpoints.find(ep => ep.type === 'bulk' && ep.direction === 'out');
      if (!input || !send) throw new Error('bulk');
      stage = 'claim';
      output.value = '장치 열기 완료 · USB 입력 인터페이스(MI_00) 연결 중';
      await device.claimInterface(0);
      if (iface.alternate.alternateSetting !== alt.alternateSetting) await device.selectAlternateInterface(0, alt.alternateSetting);
      if (active !== session) return;
      stage = 'transfer';
      button.disabled = false;
      button.textContent = 'RC 2 USB 해제';
      output.value = 'USB 인터페이스 연결 · 스틱 입력 응답을 기다리는 중';
      const started = performance.now();
      session.watchdog = setInterval(() => {
        if (performance.now() - (session.lastInput || started) < (session.lastInput ? 1000 : 4000)) return;
        finish(session, session.lastInput ? 'RC 2 입력이 끊겼습니다. 다시 연결하세요.' : 'USB 장치에서 스틱 응답을 받지 못했습니다. Assistant 2를 완전히 종료한 뒤 다시 연결하세요.');
      }, 150);
      async function write(packet) {
        const result = await device.transferOut(send.endpointNumber, packet);
        if (result.status !== 'ok' || result.bytesWritten !== packet.length) throw new Error('write');
      }
      async function poll() {
        let enabledAt = -Infinity;
        while (active === session) {
          if (performance.now() - enabledAt >= 3000) {
            await write(enableSimulator);
            enabledAt = performance.now();
          } else {
            await write(readSticks);
          }
          await new Promise(resolve => setTimeout(resolve, 20));
        }
      }
      async function receive() {
        while (active === session) {
          const reply = await device.transferIn(input.endpointNumber, 512);
          if (active !== session) return;
          if (reply.status !== 'ok') throw new Error('read');
          if (reply.data?.byteLength) readAxes(session, reply.data);
        }
      }
      await Promise.all([poll(), receive()]);
    } catch (error) {
      if (active !== session) return;
      forgetRememberedDevice();
      const code = typeof error?.name === 'string' && /^[A-Za-z]+Error$/.test(error.name) ? ' (' + error.name + ')' : '';
      const messages = {
        select: error.name === 'NotFoundError' ? 'USB 장치 선택이 완료되지 않았습니다. 목록에 장치가 있으면 이름 줄을 클릭한 뒤 연결을 누르세요.' : 'USB 장치 선택이 차단됐습니다. Windows Edge 또는 Chrome에서 이 페이지를 직접 열어 주세요.',
        open: 'USB 장치 열기 실패' + code + ' · 다른 TSD 탭과 Assistant 2를 닫고, USB를 뺐다가 다시 연결하세요.',
        configuration: 'USB 장치는 열렸지만 구성을 읽거나 선택하지 못했습니다' + code + ' · USB를 다시 연결한 뒤 장치를 선택하세요.',
        interface: 'RC 2의 USB 입력 인터페이스(MI_00)를 찾지 못했습니다.',
        claim: 'USB 입력 인터페이스(MI_00) 연결 실패' + code + ' · 다른 TSD 탭과 Assistant 2를 닫으세요. 계속되면 Windows에서 MI_00의 WinUSB 상태를 확인하세요.',
        transfer: 'RC 2 USB 통신이 끊겼습니다' + code + ' · 다시 연결하세요.'
      };
      finish(session, messages[stage]);
    }
  }

  function setup(connectButton, status, onInputChange) {
    button = connectButton;
    output = status;
    onChange = onInputChange;
    if (!navigator.usb || document.permissionsPolicy?.allowsFeature('usb') === false || document.featurePolicy?.allowsFeature('usb') === false) {
      button.disabled = true;
      output.value = 'RC 2 USB 연결을 사용하려면 Windows Edge 또는 Chrome에서 이 페이지를 직접 여세요.';
      return;
    }
    button.onclick = connect;
    refreshPermission();
    navigator.usb.addEventListener('connect', refreshPermission);
    navigator.usb.addEventListener('disconnect', event => {
      if (event.device.vendorId === filter.vendorId && event.device.productId === filter.productId) forgetRememberedDevice();
      if (active?.device === event.device) finish(active, 'RC 2 USB 케이블 연결 해제');
    });
    window.addEventListener('pagehide', () => {
      if (active) finish(active, 'RC 2 USB 연결 해제');
    });
  }

  function pad() {
    if (!active?.axes || performance.now() - active.lastInput > 250) return undefined;
    return {id, index, axes: active.axes, buttons: [], rc2Usb: true, zoomDetected: Boolean(active.zoomDetected)};
  }
  return {setup, pad, index};
})();

function formatDistance(meters) {
  if(coordinateSettings.distance==='imperial'){
    return meters>=1609.344?(meters/1609.344).toFixed(2)+' mi':Math.round(meters/.3048)+' ft';
  }
  return meters>=1000?(meters/1000).toFixed(2)+' km':Math.round(meters)+' m';

function formatDms(value,positive,negative) {
  const hundredths=Math.round(Math.abs(value)*360000);
  const d=Math.floor(hundredths/360000);
  const minutes=Math.floor(hundredths/6000)%60;
  const seconds=((hundredths%6000)/100).toFixed(2).padStart(5,'0');
  return d+'° '+String(minutes).padStart(2,'0')+'′ '+seconds+'″ '+(value<0?negative:positive);
}

function toWgs84Utm(point) {
  // Transverse Mercator series for WGS84 in the simulator's Korean coverage.
  const zone=Math.floor((point.lng+180)/6)+1;
  const lat=point.lat*degrees, sin=Math.sin(lat), cos=Math.cos(lat);
  const a=6378137, e2=.0066943799901413165, ep2=e2/(1-e2), scale=.9996;
  const n=a/Math.sqrt(1-e2*sin*sin), t=Math.tan(lat)**2, c=ep2*cos*cos;
  const x=(point.lng-(zone*6-183))*degrees*cos;
  const m=a*((1-e2/4-3*e2**2/64-5*e2**3/256)*lat
    -(3*e2/8+3*e2**2/32+45*e2**3/1024)*Math.sin(2*lat)
    +(15*e2**2/256+45*e2**3/1024)*Math.sin(4*lat)
    -35*e2**3/3072*Math.sin(6*lat));
  const easting=500000+scale*n*(x+(1-t+c)*x**3/6+(5-18*t+t*t+72*c-58*ep2)*x**5/120);
  const northing=scale*(m+n*Math.tan(lat)*(x*x/2+(5-t+9*c+4*c*c)*x**4/24
    +(61-58*t+t*t+600*c-330*ep2)*x**6/720));
  return {zone,easting,northing};
}

function formatCoordinate(point) {
  if(!point||!Number.isFinite(point.lat)||!Number.isFinite(point.lng))return '';
  if(coordinateSettings.format==='dms')return formatDms(point.lat,'N','S')+' / '+formatDms(point.lng,'E','W');
  if(coordinateSettings.format==='utm'){
    const {zone,easting,northing}=toWgs84Utm(point);
    return zone+'N · E '+easting.toFixed(1)+' · N '+northing.toFixed(1)+' m';
  }
  return point.lat.toFixed(6)+'°, '+point.lng.toFixed(6)+'°';
}

  }
}

}

function motorStart() {
  if (!['off','direct'].includes(flight.phase)||pendingDepartureView||flight.assisted||missionActive()||mission.state==='preparing') return false;
  const map = maps[selectedIndex()];
  if (!map || !terrainOk[selectedIndex()]) { setStatusMessage('DEM 로딩 후 시동할 수 있습니다'); return false; }
  const point=flight.phase==='direct'?flight.point:groundLaunchPoint(map);
  if (!validCoordinates(point.lng,point.lat)) { setStatusMessage('이륙 위치를 국내 지도 범위 안에서 선택하세요'); return false; }
  const elevation=directGround(map,point);
  if (!Number.isFinite(elevation)) { setStatusMessage('출발점 지형 로딩 후 시동할 수 있습니다'); return false; }
  Object.assign(flight,{phase:'idle',point,home:{lng:point.lng,lat:point.lat},takeoffGround:elevation,ground:elevation,alt:elevation,yaw:map.getBearing(),stopHold:0});
  directControl=false;
  routePreviewLaunch=null;
  maps.forEach(map=>browsedLaunchMaps.delete(map));
  recenterAssistMap();
  setStatusMessage('모터 시동 · 현재 위치에서 스로틀을 올려 이륙');
  announceFlight('모터가 시동되었습니다. 홈 포인트가 기록되었습니다.');
  updateFlightUI();
  return true;
}

function rememberDirectBrowse() {
  if(directBrowseView||flight.phase!=='direct'||!flight.point)return;
  directBrowseView=maps.map(map=>({tilt:map.getPitch()-90,zoom:map.getZoom()}));
  pendingSensorFrame=null;
  flight.vs=flight.speed=0;
  directFrameAt=0;
}

function resumeDirectBrowse() {
  if(!directBrowseView)return true;
  if(syncFrame){cancelAnimationFrame(syncFrame);syncFrame=0;}
  const wasSyncing=syncing;
  syncing=true;
  try {maps.forEach(map=>map.stop());} finally {syncing=wasSyncing;}
  if(!aircraftCamera())return false;
  scanAim=null;
  return true;
}

function aircraftCamera({tilt,zoom,sharedTilt=false}={}) {
  if (!flight.point) return false;
  const i=selectedIndex(), m=maps[i];
  if(!m)return false;
  const position=aircraftLens(), saved=directBrowseView;
  if(!position)return false;
  const activeTilt=tilt??saved?.[i]?.tilt??m.getPitch()-90;
  const agl=Math.max(2,flight.alt-flight.ground);
  const groundLaunch=missionActive()&&mission.run?.groundLaunch;
  // Hold the virtual focus on/near terrain instead of inheriting the ground preview's arbitrary zoom.
  const projectedRange=agl/Math.max(.25,Math.cos((90+activeTilt)*degrees));
  const launchZoom=clamp(zoomForCameraRange(m,projectedRange,position.lat),5,21);
  const activeZoom=zoom??saved?.[i]?.zoom??(groundLaunch?launchZoom:m.getZoom());
  let applied;
  if(linked()){
    const range=cameraRange(m,activeZoom,position.lat);
    applied=applySensorView(position,activeTilt,flight.yaw,map=>zoomForCameraRange(map,range,position.lat));
  } else {
    const other=maps[1-i], otherSaved=saved?.[1-i];
    if(isVisible(1-i))applySensorView(position,sharedTilt?activeTilt:otherSaved?.tilt??other.getPitch()-90,
      flight.yaw,otherSaved?.zoom??(groundLaunch?clamp(zoomForCameraRange(other,projectedRange,position.lat),5,21):other.getZoom()),[1-i]);
    applied=applySensorView(position,activeTilt,flight.yaw,activeZoom,[i]);
  }
  if(applied)directBrowseView=null;
  return applied;
}

}
async function assistFlight(mode, missionEnd=false) {
  if(mode==='return'&&(flight.assisted?.mode==='return'||flight.assisted?.home)){
    announceFlight('홈 복귀를 취소합니다.',{interrupt:true});
    cancelAssistedFlight('홈 복귀 취소 · 제자리 대기');return;
  }
  if(mode==='takeoff'&&flight.phase==='off')motorStart();
  if(!flight.point||flight.phase==='direct'||flight.assisted||mission.state==='preparing')return;
  const returnHeight=$('rthAltitude');
  if(mode==='return'&&(!flight.home||!returnHeight.validity.valid||returnHeight.value==='')){
    if(flight.home){$('flightStatusOpen').click();returnHeight.reportValidity();}
    return;
  }
  if(missionActive()&&!missionEnd){
    if(mode!=='return')return;
    finishMission('cancelled','임무 종료 · 홈 복귀로 전환');
  }
  const job={mode,ready:mode!=='return',height:12,tiles:null,home:mode==='return',missionEnd};
  const configuredReturnHeight=Number(returnHeight.value);
  flight.assisted=job;
  attitudeVelocity={forward:0,right:0};
  flight.speed=flight.vs=0;
  updateFlightUI();
  if(mode==='takeoff'){
    flight.phase='air';job.height=12;
    setStatusMessage('자동 이륙 · 지면 위 12 m');
    announceFlight('자동 이륙을 시작합니다.');
  } else if(mode==='return') {
    setStatusMessage('홈 복귀 경로의 지형을 확인하는 중');
    try {
      const from=flight.point,home=flight.home;
      job.tiles=await prepareMissionTerrain([from,home]);
      if(flight.assisted!==job)return;
      const homeGround=missionGround(home,job.tiles);
      const count=Math.ceil(geoDistance(from,home)/20);
      const elevations=Array.from({length:count+1},(_,i)=>{
        const f=count?i/count:0;
        return missionGround({lng:from.lng+(home.lng-from.lng)*f,lat:from.lat+(home.lat-from.lat)*f},job.tiles);
      });
      job.cruise=Math.max(flight.alt,homeGround+configuredReturnHeight,Math.max(...elevations)+12);
      if(!Number.isFinite(job.cruise))throw Error('RTH terrain');
      job.ready=true;
      setStatusMessage('RTH · 복귀 고도로 상승');
      announceFlight('홈 포인트로 복귀합니다.');
    } catch {
      if(flight.assisted===job){
        announceFlight('복귀 경로의 지형을 확인할 수 없습니다.',{interrupt:true});
        cancelAssistedFlight('복귀 경로의 지형 확인 실패 · 제자리 대기');
      }
    }
  } else {
    setStatusMessage('현재 위치에 착륙합니다');
    announceFlight('현재 위치에 착륙합니다.');
  }
  updateFlightUI();
}
function cancelAssistedFlight(message) {
  const ending=flight.assisted?.missionEnd;
  flight.assisted=null;flight.speed=flight.vs=0;
  if(ending)finishMission('completed',message);
  else {setStatusMessage(message);updateFlightUI();}
}
function advanceAssistedFlight(dt,now,axes) {
  const job=flight.assisted;
  if(axes.some(value=>Math.abs(value)>.2)){
    cancelAssistedFlight('스틱 입력 · 수동 비행');return;
  }
  if(!job.ready||dt<=0||document.hidden)return;
  if(job.mode==='takeoff'){
    missionAltitude(flight.ground+job.height,dt,4);
    if(Math.abs(flight.alt-flight.ground-job.height)<.01){flight.assisted=null;flight.vs=0;setStatusMessage('이륙 완료 · 현재 위치 대기');announceFlight('이륙이 완료되었습니다.');}
  } else if(job.mode==='land') {
    missionAltitude(flight.ground,dt,3);
    if(flight.alt<=flight.ground+.01){
      flight.phase='idle';
      announceFlight('착륙이 완료되었습니다.');
      cancelAssistedFlight(job.home?'홈포인트 착륙 완료':'현재 위치 착륙 완료');
    }
  } else {
    const from=flight.point, home=flight.home, remaining=geoDistance(from,home);
    const ratio=remaining>0?Math.min(1,8*dt/remaining):1;
    const point={lng:from.lng+(home.lng-from.lng)*ratio,lat:from.lat+(home.lat-from.lat)*ratio};
    const ground=missionGround(point,job.tiles);
    if(!Number.isFinite(ground)){cancelAssistedFlight('지형 확인 실패 · 현재 위치 대기');return;}
    missionAltitude(job.cruise,dt,5);
    if(flight.alt>=job.cruise-.01){
      if(!job.travelling){job.travelling=true;setStatusMessage('RTH · 홈포인트로 이동');}
      flight.ground=ground;flight.point=point;flight.speed=remaining*ratio/dt;
      if(remaining>.03){flight.course=bearingTo(from,home);turnToward(flight.course,dt);}
      if(ratio===1){
        job.mode='land';flight.speed=0;
        aircraftCamera({tilt:-90,sharedTilt:true});
        setStatusMessage('RTH · 홈포인트 도착, 직하 보기 · 착륙 중');
        announceFlight('홈 포인트에 도착했습니다. 착륙합니다.');
      }
    } else flight.speed=0;
  }
  if(now-lastFlightPaint>=65){lastFlightPaint=now;aircraftCamera();}
}

// Rev2a: finite normalized inputs and bounded horizontal stick vector (training model).
const stickDeadzone=value=>Number.isFinite(value)&&Math.abs(value)>.08?clamp(value,-1,1):0;
const safeFlightAxes=axes=>Array.from({length:4},(_,i)=>Number.isFinite(axes?.[i])?clamp(axes[i],-1,1):0);
function horizontalStickVelocity(axes,preset){
  const forward=-stickDeadzone(axes[3]),right=stickDeadzone(axes[2]);
  const scale=preset.speed/Math.max(1,Math.hypot(forward,right));
  return {forward:forward*scale,right:right*scale};
}

function directGround(map,point) {
  if(!validCoordinates(point.lng,point.lat))return null;
  const sampled=directTerrainTiles?missionGround(point,directTerrainTiles):null;
  return Number.isFinite(sampled)?sampled:renderedGround(map,point);
}

function renderedGround(map,point){
  if(!validCoordinates(point.lng,point.lat))return null;
  const terrain=map.terrain;
  if(!terrain)return null;
  const position=maplibregl.LngLat.convert(point);
  const z=Math.min(map.transform.tileZoom,displayDemMaxZoom);
  const {tileID}=terrain._getOverscaledTileIDFromLngLatZoom(position,z);
  const source=terrain.tileManager.getSourceTile(tileID,true);
  if(!source?.dem)return null;
  const {dem}=source,tile=source.tileID.canonical;
  const mercator=maplibregl.MercatorCoordinate.fromLngLat(position);
  const scale=2**tile.z,u=mercator.x*scale-tile.x,v=mercator.y*scale-tile.y;
  if(u<0||u>=1||v<0||v>=1)return null;
  // Use the source tile coordinates so parent tiles sample the correct location at high zoom.
  const px=u*dem.dim-.5,py=v*dem.dim-.5,x=Math.floor(px),y=Math.floor(py);
  const fx=px-x,fy=py-y;
  const top=dem.get(x,y)*(1-fx)+dem.get(x+1,y)*fx;
  const bottom=dem.get(x,y+1)*(1-fx)+dem.get(x+1,y+1)*fx;
  return top*(1-fy)+bottom*fy;
}

function requestDirectTerrain(point) {
  const pixel=missionDemPixel(point),key=missionDemKey(pixel.x,pixel.y);
  if(directTerrainRequest?.pending||
     (directTerrainRequest?.key===key&&performance.now()-directTerrainRequest.requestedAt<4000))return;
  const request={key,pending:true,requestedAt:performance.now()};
  directTerrainRequest=request;
  prepareMissionTerrain([point]).then(tiles=>{
    if(directTerrainRequest===request)directTerrainTiles=tiles;
  }).catch(()=>{}).finally(()=>{request.pending=false;request.requestedAt=performance.now();});
}

function flyDirect(dt,now,virtual,yawControl=virtual[0]) {
  if(!Number.isFinite(dt)||dt<=0||!Number.isFinite(now))return;
  const axes=safeFlightAxes(virtual).map(stickDeadzone);
  if(!axes.some(Boolean)){
    flight.vs=flight.speed=0;
    directFrameAt=0;
    return;
  }
  if(pendingDepartureView||dt<=0||(directFrameAt&&now-directFrameAt<30))return;
  const step=directFrameAt?Math.min(.08,(now-directFrameAt)/1000):dt;
  directFrameAt=now;
  const i=selectedIndex(),map=maps[i];
  if(!map?.getTerrain())return;
  if(flight.phase==='off'){
    const point=map.getCenter();
    if(!validCoordinates(point.lng,point.lat))return;
    departurePoint={lng:point.lng,lat:point.lat,pitch:clamp(map.getPitch(),0,65),bearing:map.getBearing(),name:'선택 지점'};
    showDepartureView();
    if(flight.phase!=='direct')return;
  }
  if(!resumeDirectBrowse())return;
  const eye=flight.point,ground=flight.ground, altitude=flight.alt;
  const yaw=(flight.yaw+stickDeadzone(yawControl)*60*step+360)%360;
  const heading=yaw*degrees,preset=flightModePresets[currentFlightMode()]||flightModePresets.N;
  const {forward,right}=horizontalStickVelocity(axes,preset);
  const north=forward*Math.cos(heading)-right*Math.sin(heading);
  const east=forward*Math.sin(heading)+right*Math.cos(heading);
  const moving=Boolean(forward||right);
  const next=moving?offsetPoint(eye,east*step,north*step):eye;
  const nextGround=moving?directGround(map,next):ground;
  if(!Number.isFinite(nextGround)&&validCoordinates(next.lng,next.lat))requestDirectTerrain(next);
  const targetAlt=altitude-axes[1]*preset.climb*step;
  const canMove=Number.isFinite(nextGround)&&targetAlt>=nextGround+directClearance-.75;
  const position=canMove?next:eye, elevation=canMove?nextGround:ground;
  const alt=Math.max(targetAlt,elevation+directClearance);
  const speed=canMove?Math.hypot(forward,right):0;
  const tilt=map.getPitch()-90,zoom=map.getZoom();
  const range=cameraRange(map,zoom,position.lat);
  const matchingZoom=view=>clamp(18+Math.log2(cameraRange(view,18,position.lat)/range),5,21);
  const views=linked()?visibleSensorIndices():[i];
  if(!applySensorView({...position,alt},tilt,yaw,matchingZoom,views)){
    flight.speed=flight.vs=0;
    return;
  }
  if(!linked()&&isVisible(1-i)){
    const other=maps[1-i];
    applySensorView({...position,alt},other.getPitch()-90,yaw,other.getZoom(),[1-i]);
  }
  if(speed||Math.abs(alt-altitude)>1e-5||Math.abs(axes[0])>0)scanAim=null;
  if(moving&&!canMove){
    showCameraRecovery(!Number.isFinite(nextGround)?'이동 방향의 고도 타일 확인 중'
      :'앞쪽 지형과 간격을 확보한 뒤 이동하세요');
  }
  Object.assign(flight,{point:{lng:position.lng,lat:position.lat},yaw,alt,ground:elevation,
    vs:(alt-altitude)/step,speed,course:speed?Math.atan2(east,north)/degrees:flight.course});
}

function flySticks(dt, now, axes, centeredThrottle, yawControl=axes[0]) {
  if(!Number.isFinite(dt)||dt<=0||!Number.isFinite(now))return;
  axes=safeFlightAxes(axes);
  const csc = cscPattern(axes);
  const throttle = -axes[1];
  if (flight.phase==='off' && csc && !flight.csc) {
    flight.cscHold += dt;
    if (flight.cscHold>=.15 && motorStart()) flight.csc = true;
  }
  if (!csc) {flight.csc=false;flight.cscHold=0;}
  if (flight.phase==='off') return;
  if (flight.phase==='idle') {
    if (!csc && throttle>.16) {
      flight.phase='air';
      setStatusMessage('수동 비행');
      announceFlight('수동 이륙을 시작합니다.');
    } else {
      flight.stopHold=throttle<-.8 && !csc && centeredThrottle ? flight.stopHold+dt : 0;
      flight.stopCscHold=csc && !flight.csc ? flight.stopCscHold+dt : 0;
      if (flight.stopHold>=1 || flight.stopCscHold>=2) {
        resetFlight();
        setStatusMessage('모터 정지');
        announceFlight('모터가 정지되었습니다.');
        if(csc){flight.csc=true;flight.cscHold=0;}
      }
    }
    if (flight.phase!=='air') return;
  }
  const dead=stickDeadzone;
  const yaw=dead(yawControl)*60*dt;
  flight.yaw=(flight.yaw+yaw+360)%360;
  const direction=flight.yaw*degrees,mode=currentFlightMode(),preset=flightModePresets[mode]||flightModePresets.N;
  const target=horizontalStickVelocity(axes,preset);
  const oldSpeed=flight.speed,oldVs=flight.vs,oldAlt=flight.alt;
  // Educational velocity response; not aircraft-measured acceleration coefficients.
  const active=Boolean(target.forward||target.right);
  const responseRate=mode==='A'?(active?2.5:.4):(active?4:6);
  const response=mode==='A'?Math.min(1,dt*responseRate):1-Math.exp(-dt*responseRate);
  attitudeVelocity.forward+=(target.forward-attitudeVelocity.forward)*response;
  attitudeVelocity.right+=(target.right-attitudeVelocity.right)*response;
  if(!active&&Math.hypot(attitudeVelocity.forward,attitudeVelocity.right)<.02){
    attitudeVelocity.forward=attitudeVelocity.right=0;
  }
  const bounded=Math.max(1,Math.hypot(attitudeVelocity.forward,attitudeVelocity.right)/preset.speed);
  attitudeVelocity.forward/=bounded;attitudeVelocity.right/=bounded;
  const forward=attitudeVelocity.forward,right=attitudeVelocity.right;
  const north=forward*Math.cos(direction)-right*Math.sin(direction);
  const east=forward*Math.sin(direction)+right*Math.cos(direction);
  const commandedVs=dead(throttle)*preset.climb;
  const intendedAlt=oldAlt+commandedVs*dt;
  let elevation=flight.ground;
  let speed=Math.hypot(forward,right);
  if(speed>0){
    const next=offsetPoint(flight.point,east*dt,north*dt),map=maps[selectedIndex()];
    let ground=null;
    if(map&&validCoordinates(next.lng,next.lat)){
      try{ground=directGround(map,next);}catch(error){ground=null;}
      if(!Number.isFinite(ground)){
        try{requestDirectTerrain(next);}catch(error){}
      }
    }
    if(Number.isFinite(ground)&&Math.min(oldAlt,intendedAlt)>=ground){
      flight.point=new maplibregl.LngLat(next.lng,next.lat);
      elevation=ground;
      flight.course=Math.atan2(east,north)/degrees;
    }else{
      // No position/altitude teleport on a missing or obstructed terrain sample.
      speed=0;attitudeVelocity.forward=attitudeVelocity.right=0;
    }
  }
  flight.ground=elevation;
  flight.speed=speed;
  // Simulator-only manual altitude: no artificial 300 m ceiling; retain the existing ground floor.
  flight.alt=Math.max(intendedAlt,Math.min(elevation,oldAlt));
  flight.vs=(flight.alt-oldAlt)/dt;
  if(flight.alt<=elevation&&commandedVs<0){
    flight.phase='idle';flight.vs=flight.speed=0;
    attitudeVelocity.forward=attitudeVelocity.right=0;
    setStatusMessage('착륙 · 모터 회전 중');
    announceFlight('착륙했습니다. 모터가 회전 중입니다.');
  }
  if(now-lastFlightPaint>=50&&(yaw||flight.vs||flight.speed||oldSpeed||oldVs||flight.phase==='idle')){
    lastFlightPaint=now;
    aircraftCamera();
  }
}

function bindPanelDrag(handle, onStart) {
  let drag=null;
  handle.addEventListener('pointerdown',event=>{
    if(event.button!==0||drag)return;
    const update=onStart(event);
    if(!update)return;
    handle.setPointerCapture(event.pointerId);
    drag={id:event.pointerId,x:event.clientX,y:event.clientY,update};
  });
  handle.addEventListener('pointermove',event=>{
    if(drag?.id!==event.pointerId)return;
    const dx=event.clientX-drag.x,dy=event.clientY-drag.y;
    if(drag.moved||Math.hypot(dx,dy)>=5){
      drag.moved=true;
      drag.update(dx,dy);
    }
  });
  const finish=event=>{if(drag?.id===event.pointerId)drag=null;};
  for(const name of ['pointerup','pointercancel','lostpointercapture'])handle.addEventListener(name,finish);
}

function bindPanelKeys(handle, move, reset) {
  handle.addEventListener('keydown',event=>{
    const directions={ArrowLeft:[-1,0],ArrowRight:[1,0],ArrowUp:[0,-1],ArrowDown:[0,1]};
    const direction=directions[event.key];
    if(!direction&&event.key!=='Home')return;
    event.preventDefault();
    event.stopPropagation();
    if(direction){
      const step=event.shiftKey?36:12;
      move(direction[0]*step,direction[1]*step);
    } else reset();
  });
}

function resizeStageSensors() {
  const indices=visibleSensorIndices();
  const resized=indices.filter(i=>{
    const map=maps[i], frame=map.getContainer();
    return map.transform.width!==frame.clientWidth||map.transform.height!==frame.clientHeight;
  });
  if(!resized.length)return;
  const selected=selectedIndex();
  const anchors=linked()
    ? [{index:selected, affected:indices, anchor:captureCameraAnchor(maps[selected])}]
    : resized.map(index=>({index,affected:[index],anchor:captureCameraAnchor(maps[index])}));
  viewRevision++;
  pendingSensorFrame=null;
  if(syncFrame){cancelAnimationFrame(syncFrame);syncFrame=0;}
  const wasSyncing=syncing;
  syncing=true;
  try {resized.forEach(index=>maps[index].resize());}
  finally {syncing=wasSyncing;}
  anchors.forEach(({anchor,index,affected})=>restoreResizedCamera(anchor,index,affected,false));
  updateCameraUI();
  indices.forEach(i=>maps[i].triggerRepaint());
}

function setupAssistLayout() {
  const stage=$('stage');
  const cards=[
    {panel:$('miniMapPanel'),body:$('miniMapBody'),grip:$('miniMapGrip'),kind:'mini',side:'left'},
    {panel:$('visionPanel'),body:$('visionBody'),grip:$('visionGrip'),kind:'vision',side:'right'}
  ];
  function bounds(card) {
    // Reserve room for the title and controls, independent of the image height.
    const max=Math.max(80,Math.min(520,stage.clientWidth-16,Math.floor((stage.clientHeight-70)*16/9)));
    return {min:Math.min(max,112),max};
  }
  function front(card) {
    for(const entry of cards)entry.panel.toggleAttribute('data-front',entry===card);
  }
  function layout(card) {
    const {panel,body}=card;
    const expanded=panel.dataset.size==='large',closed=body.hidden;
    const limit=bounds(card);
    panel.style.width=!expanded&&!closed&&card.width?clamp(card.width,limit.min,limit.max)+'px':'';
    const xMax=Math.max(0,stage.clientWidth-panel.offsetWidth-4);
    const yMax=Math.max(0,stage.clientHeight-panel.offsetHeight-4);
    if(card.position&&!expanded){
      panel.style.left=clamp(card.position.x*stage.clientWidth,0,xMax)+'px';
      panel.style.top=clamp(card.position.y*stage.clientHeight,0,yMax)+'px';
      panel.style.right='auto';
      panel.style.bottom='auto';
    } else {
      for(const name of ['left','right','top','bottom'])panel.style.removeProperty(name);
    }
    if(!expanded&&!closed){
      card.minus.disabled=panel.offsetWidth<=limit.min+1;
      card.plus.disabled=panel.offsetWidth>=limit.max-1;
    }
  }
  const observer=new ResizeObserver(entries=>{
    for(const {target} of entries){
      if(target===stage){cards.forEach(layout);continue;}
      for(const card of cards){
        if(target===card.panel){layout(card);continue;}
        if(target===card.body&&!target.hidden&&target.clientWidth&&target.clientHeight)flightAssist[card.kind]?.resize?.();
      }
    }
  });
  for(const card of cards){
    const {panel,body,grip}=card;
    card.minus=panel.querySelector('[data-inset-step="-1"]');
    card.plus=panel.querySelector('[data-inset-step="1"]');
    const put=(x,y)=>{
      if(panel.dataset.size==='large')return;
      const rect=stage.getBoundingClientRect();
      if(!rect.width||!rect.height)return;
      card.position={
        x:clamp(x-rect.left,0,Math.max(0,rect.width-panel.offsetWidth-4))/rect.width,
        y:clamp(y-rect.top,0,Math.max(0,rect.height-panel.offsetHeight-4))/rect.height
      };
      front(card);
      layout(card);
    };
    const resize=(width,keepRight)=>{
      if(body.hidden||panel.dataset.size==='large')return;
      const limit=bounds(card);
      const rect=panel.getBoundingClientRect();
      card.width=clamp(Math.round(width),limit.min,limit.max);
      front(card);
      layout(card);
      if(keepRight&&card.position)put(rect.right-panel.offsetWidth,rect.top);
    };
    const reset=()=>{
      card.position=null;card.width=null;
      layout(card);
    };
    bindPanelDrag(grip,()=>{
      if(panel.dataset.size==='large')return;
      const rect=panel.getBoundingClientRect();
      return (dx,dy)=>put(rect.left+dx,rect.top+dy);
    });
    bindPanelKeys(grip,(dx,dy)=>{
      const rect=panel.getBoundingClientRect();
      put(rect.left+dx,rect.top+dy);
    },reset);
    grip.addEventListener('dblclick',reset);
    const handle=panel.querySelector('.insetResize');
    const step=()=>Math.max(24,Math.round(stage.clientWidth*.045));
    bindPanelDrag(handle,()=>{
      const width=panel.offsetWidth;
      const direction=card.side==='right'?-1:1;
      return dx=>resize(width+dx*direction,direction<0);
    });
    bindPanelKeys(handle,dx=>resize(panel.offsetWidth+dx*step(),card.side==='right'),reset);
    card.minus.onclick=()=>resize(panel.offsetWidth-step(),card.side==='right');
    card.plus.onclick=()=>resize(panel.offsetWidth+step(),card.side==='right');
    panel.addEventListener('pointerdown',()=>front(card));
    for(const target of [body,panel])observer.observe(target);
    // Fullscreen and collapse change display and width immediately.
    new MutationObserver(()=>layout(card)).observe(panel,{attributes:true,attributeFilter:['data-size','hidden']});
    layout(card);
  }
  observer.observe(stage);
}

function virtualAxes() {

    mouseAxes[0] || Number(keyState.has('d'))-Number(keyState.has('a')),
    mouseAxes[1] || Number(keyState.has('s'))-Number(keyState.has('w')),
    mouseAxes[2] || Number(keyState.has('arrowright'))-Number(keyState.has('arrowleft')),
    mouseAxes[3] || Number(keyState.has('arrowdown'))-Number(keyState.has('arrowup'))
  ];
}
function clearKeyboardControls() {
  keyState.clear();
  Object.assign(keyboardYaw,{direction:0,held:0,axis:0});
  for(const control of keyboardCamera)Object.assign(control,{direction:0,delay:0,elapsed:0});
}
function advanceKeyboardYaw(dt) {
  const direction=Number(keyState.has('d'))-Number(keyState.has('a'));
  if(!direction||direction!==keyboardYaw.direction)keyboardYaw.held=0;
  keyboardYaw.direction=direction;
  if(!direction){keyboardYaw.axis=0;return 0;}
  const t=Math.min(1,keyboardYaw.held/1.4);
  keyboardYaw.axis=direction*(.1+.35*t*t*(3-2*t));
  keyboardYaw.held+=dt;
  return keyboardYaw.axis;
}
function stepKeyboardTilt(delta) {
  if(!resumeDirectBrowse())return false;
  const map=maps[selectedIndex()];
  if(!map)return false;
  const tilt=map.getPitch()-90, next=clamp(tilt+delta,-90,35);
  return Math.abs(next-tilt)>.0001 && setGimbal(next,false);
}
function cameraKeyDirection(control) {
  return Number(keyState.has(control.more))-Number(keyState.has(control.less));
}
function advanceKeyboardCamera(dt) {
  for(const control of keyboardCamera){
    const direction=cameraKeyDirection(control);
    if(direction!==control.direction)Object.assign(control,{direction,delay:0,elapsed:0});
    if(!direction)continue;
    control.elapsed+=Math.max(0,dt-control.delay);
    control.delay=Math.max(0,control.delay-dt);
    if(control.elapsed>0&&control.elapsed>=control.interval){
      control.apply(direction*control.rate*control.elapsed);
      control.elapsed=0;
    }
  }
}
function installVirtualSticks() {
  for (const [id,index] of [['leftStick',0],['rightStick',2]]) {
    const stick=$(id);
    const read = e => {
      const box=stick.getBoundingClientRect();
      mouseAxes[index]=clamp((e.clientX-box.left-box.width/2)/(box.width*.39),-1,1);
      mouseAxes[index+1]=clamp((e.clientY-box.top-box.height/2)/(box.height*.39),-1,1);
    };
    let pointer=null;
    stick.onpointerdown=e=>{
      if(pointer!==null||e.button!==0)return;
      pointer=e.pointerId;
      stick.setPointerCapture(pointer);
      read(e);
      if(flight.phase==='direct'&&!missionActive()&&mission.state!=='preparing'){
        flyDirect(1/60,performance.now(),virtualAxes(),mouseAxes[0]||keyboardYaw.axis);
      }
    };
    stick.onpointermove=e=>{if(pointer===e.pointerId)read(e);};
    const release=e=>{
      if(pointer!==e.pointerId)return;
      pointer=null;
      mouseAxes[index]=0;mouseAxes[index+1]=0;
    };
    for(const type of ['pointerup','pointercancel','lostpointercapture'])stick.addEventListener(type,release);
  }
  $('directControl').onclick=()=>{
    directControl=!directControl;
    pendingDepartureView=null;
    if(flight.phase==='direct')resetFlight();
    mouseAxes.fill(0);
    clearKeyboardControls();
    pendingSensorFrame=null;
    showCameraRecovery();
    if(directControl){
      const map=maps[selectedIndex()],point=map?.getCenter();
      if(point&&validCoordinates(point.lng,point.lat)){
        departurePoint={lng:point.lng,lat:point.lat,pitch:clamp(map.getPitch(),0,65),bearing:map.getBearing(),name:'선택 지점'};
        showDepartureView();
      }
    }else setStatusMessage('수동 비행 대기 · 모터 시동 또는 자동 이륙');
    updateFlightUI();
  };
  $('virtualMotorStart').onclick=()=>{motorStart();};
  window.addEventListener('keydown',event=>{
    const key=event.key.toLowerCase(), camera=keyboardCamera.find(control=>key===control.less||key===control.more);
    if(event.altKey||event.ctrlKey||event.metaKey||event.isComposing||document.querySelector('dialog[open]')
       ||event.target.closest?.('input, select, textarea, [contenteditable], .panelGrip, .insetResize'))return;
    if(!camera&&($('flightPanel').hidden||!flightKeys.has(key)))return;
    // Consume flight keys before the focused MapLibre canvas can pan or rotate itself.
    event.preventDefault();
    event.stopImmediatePropagation();
    if(event.repeat||keyState.has(key))return;
    keyState.add(key);
    if(camera){
      const direction=cameraKeyDirection(camera);
      Object.assign(camera,{direction,delay:.16,elapsed:0});
      if(direction)camera.apply(direction*camera.tap);
    }
  },{capture:true});
  window.addEventListener('keyup',event=>keyState.delete(event.key.toLowerCase()),{capture:true});
  const release=()=>{clearKeyboardControls();mouseAxes.fill(0);};
  window.addEventListener('blur',release);
  document.addEventListener('visibilitychange',()=>{if(document.hidden)release();});
}

    };
    const invert=document.createElement('label');
    if (type==='axis' && supported) {
      const box=document.createElement('input');
      box.type='checkbox';box.checked=field.invert;
      box.onchange=()=>{field.invert=box.checked;storeRc();};
      invert.append(box,'반전');
    }
    row.append(name,value,learn,invert);
    $('rcMappings').append(row);
  }
}
function readRc(field,pad) {
  if (field.index===null) return 0;
  if (field.type==='button') return pad.buttons[field.index]?.value || 0;
  const raw=pad.axes[field.index];
  if(!Number.isFinite(raw))return 0;
  const diff=raw-field.neutral;
  return (field.invert?-1:1)*clamp(diff/Math.max(.1,diff>=0?1-field.neutral:1+field.neutral),-1,1);
}
function readController(now,dt) {
  const pad=now-rc.lastPaint>500 ? scanPads() : getPads().find(p=>p.index===rc.index);
  if(now-rc.lastPaint>500)rc.lastPaint=now;
  if (!pad || !rc.settings) {
    if (rc.index===rc2Usb.index) resumeNeutral=true;
    return null;
  }
  if($('rcDialog').open){
    const signature=pad.axes.map(v=>v.toFixed(2)).join(' / ')+' · 버튼 '+(pad.buttons.flatMap((button,i)=>button.pressed||button.value>.5?[i+1]:[]).join(', ')||'—');
    if(signature!==lastRcInput){lastRcInput=signature;$('rcSignals').value='축 '+signature;}
  }
  const input=rc.settings.input;
  if (rc.learning) {
    const {key,type,base,at}=rc.learning;
    const list=type==='axis'?pad.axes:pad.buttons.map(b=>b.value);
    const index=list.reduce((best,v,i)=>Math.abs(v-base[i])>Math.abs((list[best]??0)-(base[best]??0))?i:best,0);
    if(Math.abs((list[index]??0)-(base[index]??0))>(type==='axis'?.4:.6)) {
      for(const [other,field] of Object.entries(input)) {
        if(other!==key&&field.type===type&&field.index===index)field.index=null;
      }
      input[key].index=index;
      input[key].neutral=type==='axis'&&Math.abs(base[index])<.35?base[index]:0;
      rc.learning=null;storeRc();renderMapping();
    } else if (now-at>10000) {rc.learning=null;$('rcState').value='입력 지정 시간 초과';}
    return null;
  }
  const axes=['lx','ly','rx','ry'].map(k=>readRc(input[k],pad));
  if($('rcDialog').open) {
    rc.returnStarted=null;
    if(now-rc.deltaAt>100) {
      rc.deltaAt=now;
      $('rcState').value=pad.id+' · 스틱 '+axes.map(n=>n.toFixed(2)).join(' / ');
      for(const [key,,type] of mappingFields) {
        const field=input[key];
        if(field.index!==null && !$('rcBind-'+key).disabled)$('rcValue-'+key).value=(type==='axis'?'축 ':'버튼 ')+(field.index+1)+' · '+readRc(field,pad).toFixed(2);
      }
    }
    for(const key of Object.keys(buttonActions))rc.buttons[key]=readRc(input[key],pad)>.5;
    return null;
  }
  if ($('review').open || document.hidden || !document.hasFocus()) {
    rc.returnStarted=null;
    resumeNeutral=true;
    for(const key of Object.keys(buttonActions))rc.buttons[key]=readRc(input[key],pad)>.5;
    return null;
  }
  if(resumeNeutral) {
    const centered=[axes[0],axes[2],axes[3]].every(v=>Math.abs(v)<.15);
    if(!centered || (rc.settings.spring?Math.abs(axes[1])>=.15:axes[1]<.7)) return null;
    resumeNeutral=false;
  }
  if(now-rc.deltaAt>50) {
    const interval=Math.min(.12,Math.max(dt,(now-rc.deltaAt)/1000));
    rc.deltaAt=now;
    const tilt=readRc(input.tilt,pad);
    if(Math.abs(tilt)>.1 && maps.length) setGimbal(maps[selectedIndex()].getPitch()-90+tilt*45*interval);
    if(input.modeDial.index!==null){
      const value=readRc(input.modeDial,pad),mode=value<-.45?'F':value>.45?'S':'N';
      if(mode!==modeSwitch)setFlightMode(mode);
    }
  }
  const zoom=readRc(input.zoom,pad);
  if(Math.abs(zoom)>.1&&maps.length)zoomStep(zoom*.75*dt);
  for(const [key,action] of Object.entries(buttonActions)) {
    const pressed=readRc(input[key],pad)>.5;
    if(key==='return'){
      if(!pressed)rc.returnStarted=null;
      else if(!rc.buttons[key]){
        if(flight.assisted?.home)action();
        else rc.returnStarted=now;
      }else if(rc.returnStarted!==null&&now-rc.returnStarted>=700){
        rc.returnStarted=null;action();
      }
    }else if(pressed&&(key==='zoomIn'||key==='zoomOut')){
      if(!rc.buttons[key])action();
      else zoomStep((key==='zoomIn'?1:-1)*.75*dt);
    }else if(pressed && !rc.buttons[key]) action();
    rc.buttons[key]=pressed;
  }
  return axes;
}
const flightOptionalFailures=new Set();
function runFlightOptional(kind,action){
  try{
    const result=action();
    flightOptionalFailures.delete(kind);
    return result;
  }catch(error){
    if(!flightOptionalFailures.has(kind))console.warn('TSD frame optional: '+kind,error);
    flightOptionalFailures.add(kind);
    return undefined;
  }
}
function flightTick(now) {
  try{flightTickBody(now);}finally{requestAnimationFrame(flightTick);}
}
function flightTickBody(now) {
  if(tsdPerformance.debug||tsdPerformance.latencyProbe)runFlightOptional('diagnostics',()=>samplePerformance(now));
  const elapsed=Math.max(0,(now-lastFlightFrame)/1000);
  const dt=Math.min(.05,elapsed);
  lastFlightFrame=now;
  const dialogOpen=Boolean(document.querySelector('dialog[open]'));
  if(!document.hidden&&(!tsdLowPower||now-tsdPerformance.lastLinkAt>=1000/12)){
    tsdPerformance.lastLinkAt=now;
    runFlightOptional('link-view',()=>window.tsdEoirLink?.updateView(now));
  }
  if(!document.hidden)advanceZoomInput(dt);
  if(document.hidden||dialogOpen){if(keyState.size)clearKeyboardControls();}
  else advanceKeyboardCamera(dt);
  tickTargetMotion(now);
  const physical=readController(now,dt);
  const local=virtualAxes();
  const touch=!document.hidden&&(!dialogOpen||$('touchLinkDialog').open)?runFlightOptional('link-input',()=>window.tsdEoirLink?.frame(dt,now)):null;
  const remote=touch&&!local.some(stickDeadzone)&&touch.some(stickDeadzone);
  const virtual=remote?touch:local;
  const keyboardAxis=advanceKeyboardYaw(dt), virtualYaw=remote?touch[0]:mouseAxes[0]||keyboardAxis;
  const useVirtual=!physical || virtual.some(Boolean);
  const axes=useVirtual?virtual:physical;
  // A throttle that rests at its lower endpoint is neutral for mission override.
  const overrideAxes=useVirtual||rc.settings.spring?axes:[axes[0],axes[1]>.7?0:1-axes[1],axes[2],axes[3]];
  if(!$('flightPanel').hidden)updateSticks(useVirtual?[virtualYaw,...axes.slice(1)]:axes);
  if(flight.assisted){
    const safeAxes=dialogOpen?[]:overrideAxes;
    let remaining=Math.min(.25,elapsed);
    while(remaining>1e-6&&flight.assisted){
      const step=Math.min(.05,remaining);
      remaining-=step;
      advanceAssistedFlight(step,now,safeAxes);
    }
  } else if (missionActive()) {
    if (!dialogOpen && !document.hidden && overrideAxes.some(value=>Math.abs(value)>.3)) {
      pauseMission('스틱 입력 · 임무 일시정지');
    }
    let remaining=Math.min(.25,elapsed);
    while(remaining>1e-6&&mission.state==='running'){
      const step=Math.min(.05,remaining);
      remaining-=step;
      advanceMission(step,now,remaining<=1e-6);
    }
  } else if (mission.state!=='preparing'&&!document.hidden&&!dialogOpen) {
    if(flight.phase==='direct'||(flight.phase==='off'&&directControl&&virtual.some(stickDeadzone))){
      flyDirect(dt,now,virtual,virtualYaw);
    } else if(physical||virtual.some(Boolean)||flight.phase!=='off'){
      flySticks(dt,now,axes,useVirtual||rc.settings.spring,useVirtual?virtualYaw:axes[0]);
    }
  }
  runFlightOptional('prefetch',()=>prefetchFlightTerrain(now));
  if(now-lastFlightText>160){
    lastFlightText=now;
    runFlightOptional('flight-ui',()=>updateFlightUI());
    runFlightOptional('flight-assist',()=>updateFlightAssist(now));
    if($('zoomTest').getAttribute('aria-pressed')==='true'){
      runFlightOptional('zoom-eo',()=>updateZoomReference(0));
      runFlightOptional('zoom-ir',()=>updateZoomReference(1));
    }
  }
}

const flightAssist = {
  mini: null, vision: null, miniSize: 'small', north: true, follow: true, miniMarkers: {},
  collapsed: false, locked: null, temporary: null, expires: 0, held: false, holdTimer: 0, holdPointer: null,
  visionPose: null, visionRendered: false, miniPlan: undefined
};
const assistStyleState = new WeakMap();
// Rev5a: monitor visual rendering independently of the flight loop.
const visionHealth = new WeakMap();
let visionRetryAfter=0;
function resetVisionHealth(map){
  if(map)visionHealth.set(map,{pendingSince:null,repainted:false,failedAt:null,lostAt:null,size:null,createdAt:performance.now()});
}
function suspendVision(){
  flightAssist.visionPose=null;
  flightAssist.visionRendered=false;
  const h=visionHealth.get(flightAssist.vision);
  if(h){h.pendingSince=null;h.repainted=false;h.size=null;}
}
function recoverVisionMap(map,now){
  if(flightAssist.vision!==map)return;
  // Release the old renderer before allocating a replacement; do not touch EO/IR.
  flightAssist.vision=null;
  suspendVision();
  visionRetryAfter=now+3000;
  try{map.remove();}catch(error){
    // Do not allocate another GPU context if cleanup could not be confirmed.
    flightAssist.vision=map;
    assistStyleState.set(map,'failed');
    const h=visionHealth.get(map);if(h)h.failedAt=now;
    console.warn('TSD vision cleanup',error);
  }
  const message=$('visionStatus');message.hidden=false;
  message.value='모의 비전 재연결 대기';
}
const visionLabels = {0:'FRONT', 90:'RIGHT', 180:'BACK', 270:'LEFT'};
const assistColors = {near:'#ff545a', warning:'#ffad42', detected:'#62dba2'};
const assistRouteId = 'tsd-assist-route';

function resolveVisionDirection(yaw, course, speed, locked, temporary, expires, now) {
  if (temporary !== null && now < expires) return temporary;
  if (locked !== null) return locked;
  if (speed < .2) return 0;
  return ((Math.round((course-yaw)/90)*90)%360+360)%360;
}

function createAssistMap(kind) {
  const container = kind === 'mini' ? 'miniMapCanvas' : 'visionCanvas';
  const message = $(kind === 'mini' ? 'miniMapStatus' : 'visionStatus');
  if (!window.maplibregl) return null;
  if(kind==='vision'&&!flightAssist.vision&&performance.now()<visionRetryAfter)return null;
  if(flightAssist[kind]===false){
    if(kind!=='vision')return null;
    flightAssist.vision=null;
  }
  if (flightAssist[kind]) return flightAssist[kind];
  try {
    const map = new maplibregl.Map({
      container, style:mapStyle(), center:initialPoint, zoom:15, interactive:kind==='mini',
      attributionControl:false, pitchWithRotate:false, dragRotate:false, renderWorldCopies:false,
      maxPitch:kind==='vision'?90:0, maxTileCacheSize:tsdLowPower?10:24, fadeDuration:0,
      canvasContextAttributes:{antialias:false, powerPreference:'low-power'}
    });
    flightAssist[kind]=map;
    assistStyleState.set(map,'pending');
    if(kind==='vision')resetVisionHealth(map);
    if(tsdLowPower&&typeof map.setPixelRatio==='function')map.setPixelRatio(.7);
    if(kind==='mini'){
      map.touchZoomRotate.disableRotation();
      map.on('dragstart',()=>{flightAssist.follow=false;});
      map.on('click',e=>{if(flightAssist.miniSize==='large')pickMissionCenter(e.lngLat);});
    }
    map.on('style.load',()=>{
      if(flightAssist[kind]!==map)return;
      assistStyleState.set(map,'pending');
      message.hidden=false;
      if(kind==='vision'){suspendVision();resetVisionHealth(map);}
      try {
        if (kind==='vision') {
          if(!map.getSource('assist-dem'))map.addSource('assist-dem',{type:'raster-dem',tiles:[DEM],tileSize:256,encoding:'terrarium',maxzoom:displayDemMaxZoom,attribution:'Elevation: Mapzen/AWS Terrain Tiles'});
          if(!map.getTerrain())map.setTerrain({source:'assist-dem',exaggeration:1});
          if(tsdLowPower&&map.terrain?.meshSize===128)map.terrain.meshSize=32;
          map.setSky({'sky-color':'#8fa9bc','horizon-color':'#d5dde1','fog-color':'#d5dde1'});
          map.setCenterClampedToGround(false);
        } else {



          if(!map.getSource(assistRouteId))map.addSource(assistRouteId,{type:'geojson',data:{type:'FeatureCollection',features:[]}});
          if(!map.getLayer(assistRouteId))map.addLayer({id:assistRouteId,type:'line',source:assistRouteId,paint:{'line-color':'#84e0ae','line-width':3}});
          flightAssist.miniPlan=undefined;
        }
        assistStyleState.set(map,'ready');
      } catch(error) {
        assistStyleState.set(map,'failed');
        if(kind==='vision')visionHealth.get(map).failedAt=performance.now();
        message.value='보조 지도 초기화 실패 · 지도 전환 후 다시 시도하세요';
        console.warn(error);
      }
    });
    map.on('idle',()=>{
      if(flightAssist[kind]!==map)return;
      if(assistStyleState.get(map)!=='ready'||!map.isStyleLoaded()||!map.areTilesLoaded())return;
      if(kind==='vision')flightAssist.visionRendered=true;
      message.hidden=kind==='mini'||((flight.phase==='air'||flight.phase==='direct')&&!!flight.point);
    });
    if(kind==='vision'){
      map.on('render',()=>{
        if(flightAssist.vision!==map||assistStyleState.get(map)!=='ready')return;
        const h=visionHealth.get(map);
        if(h&&h.lostAt===null){h.pendingSince=null;h.repainted=false;}
      });
      map.on('webglcontextlost',()=>{
        if(flightAssist.vision!==map)return;
        suspendVision();visionHealth.get(map).lostAt=performance.now();
        message.value='모의 비전 렌더링 복구 대기';message.hidden=false;
      });
      map.on('webglcontextrestored',()=>{
        if(flightAssist.vision!==map)return;
        suspendVision();resetVisionHealth(map);
        map.triggerRepaint();
      });
    }
    map.on('error',()=>{if(flightAssist[kind]!==map)return;if(!message.hidden&&assistStyleState.get(map)!=='failed') message.value=kind==='mini'?'지도 타일 연결 대기':'모의 비전 지형 연결 대기';});
    return map;
  } catch {
    const partial=flightAssist[kind];
    if(kind==='vision'&&partial){
      try{partial.remove();}catch(error){
        assistStyleState.set(partial,'failed');
        if(!visionHealth.has(partial))resetVisionHealth(partial);
        visionHealth.get(partial).failedAt=performance.now();
        message.value='모의 비전 초기화 실패';message.hidden=false;
        return null;
      }
    }
    flightAssist[kind]=false;
    if(kind==='vision')visionRetryAfter=performance.now()+3000;
    message.value='WebGL을 사용할 수 없습니다';
    message.hidden=false;
    return null;
  }
}

    }
  }
}

function updateMiniMap(point, yaw) {
  if($('miniMapBody').hidden)return;
  const map=createAssistMap('mini');
  if(!map)return;
  const center=map.getCenter(), bearing=flightAssist.north?0:yaw;
  if((flightAssist.follow&&geoDistance(center,point)>.4)||Math.abs(headingDelta(map.getBearing(),bearing))>.1){
    map.jumpTo({center:flightAssist.follow?point:center,bearing});
  }
  updateInsetMarker('home',flight.home);
  updateInsetMarker('pin',observationPoint);
  updateInsetMarker('aircraft',point,yaw);
  const plan=mission.plan, source=map.getSource(assistRouteId);
  if(source&&flightAssist.miniPlan!==plan){
    source.setData({type:'FeatureCollection',features:plan?[{type:'Feature',properties:{},geometry:{type:'LineString',coordinates:plan.points.map(p=>[p.lng,p.lat])}}]:[]});
    flightAssist.miniPlan=plan;
  }
}

function updateVision(now, yaw) {
  const direction=resolveVisionDirection(yaw,flight.course,flight.speed,flightAssist.locked,flightAssist.temporary,flightAssist.expires,now);
  const heading=$('visionHeading');
  heading.textContent=visionLabels[direction]+(flightAssist.locked!==null?' · 🔒':'');

  heading.setAttribute('aria-pressed',String(flightAssist.locked!==null));
  for(const button of document.querySelectorAll('[data-vision]')){
    button.setAttribute('aria-pressed',String(Number(button.dataset.vision)===direction));
  }
  if($('visionPanel').hidden||$('visionBody').hidden){suspendVision();return;}
  const message=$('visionStatus');
  if((flight.phase!=='air'&&flight.phase!=='direct')||!flight.point){
    message.value='이륙 또는 수동 즉시 조종 후 모의 비전 표시';message.hidden=false;
    $('visionCanvas').hidden=true;

    return;
  }
  $('visionCanvas').hidden=false;
  if(flightAssist.vision !== false){
    const failed=assistStyleState.get(flightAssist.vision)==='failed';
    message.value=failed?'보조 지도 초기화 실패 · 지도 전환 후 다시 시도하세요':'모의 비전 지형 로딩 중';
    message.hidden=!failed&&flightAssist.visionRendered;
  }
  const map=createAssistMap('vision');
  if(!map){if(now<visionRetryAfter){message.value='모의 비전 재연결 대기';message.hidden=false;}return;}
  const h=visionHealth.get(map);
  if(h.lostAt!==null||assistStyleState.get(map)==='failed'){
    message.value='모의 비전 렌더링 복구 대기';message.hidden=false;
    if(now-(h.lostAt??h.failedAt??now)>=6000)recoverVisionMap(map,now);
    return;
  }
  if(assistStyleState.get(map)!=='ready'||!map.getTerrain()){
    if(now-h.createdAt>=15000)recoverVisionMap(map,now);
    return;
  }
  const box=map.getContainer();
  if(!(box.clientWidth>0&&box.clientHeight>0))return;
  const size=box.clientWidth+'|'+box.clientHeight;
  if(h.size!==size){map.resize();h.size=size;flightAssist.visionPose=null;}
  if(h.pendingSince!==null){
    const elapsed=now-h.pendingSince;
    if(elapsed>=6000){recoverVisionMap(map,now);return;}
    if(elapsed>=2000){
      flightAssist.visionRendered=false;
      message.value='모의 비전 영상 갱신 재시도 중';message.hidden=false;
      if(!h.repainted){h.repainted=true;map.triggerRepaint();}
    }
  }
  const point=flight.point, pose={lng:point.lng,lat:point.lat,alt:flight.alt};
  const zoom=Math.max(14,Math.min(17,17-Math.log2(Math.max(12,flight.alt-flight.ground)/12)));
  const signature=[pose.lng,pose.lat,pose.alt,yaw,direction,zoom].join('|');
  if(signature===flightAssist.visionPose)return;
  map.setCenterClampedToGround(false);
  // Keep the oldest unacknowledged request even while the aircraft is moving.
  if(h.pendingSince===null){h.pendingSince=now;h.repainted=false;}
  map.jumpTo(cameraAt(map,pose,-16,yaw+direction,zoom));
  flightAssist.visionPose=signature;
}

function classifyTerrainContact(distance, warning, near) {
  if(distance>16)return '';
  if(distance<=near)return 'near';
  return distance<=warning?'warning':'detected';
}

function assessTerrainContacts(map, point, altitude, yaw) {
  const contacts=[];
  for(let sector=0;sector<8;sector++){
    const angle=(yaw+sector*45)*degrees;
    let distance=Infinity, sampled=false;
    for(const m of [2,4,6,8,10,12,14,16]){
      const sample=offsetPoint(point,Math.sin(angle)*m,Math.cos(angle)*m);
      const elevation=map.queryTerrainElevation([sample.lng,sample.lat]);
      if(!Number.isFinite(elevation))continue;
      sampled=true;
      if(elevation>=altitude-.5){distance=m;break;}
    }
    contacts.push(sampled?distance:null);
  }
  return contacts;
}

function colorizeIR(data,table,brightness=1,contrast=1) {
  if (!data || data.length%4!==0 || !table || table.length!==768) return false;
  if(!Number.isFinite(brightness)||!Number.isFinite(contrast))return false;
  for (let i=0;i<data.length;i+=4) {
    const luminance=data[i]*.2126+data[i+1]*.7152+data[i+2]*.0722;
    const index=3*clamp(Math.round((luminance*brightness-128)*contrast+128),0,255);
    data[i]=table[index];
    data[i+1]=table[index+1];
    data[i+2]=table[index+2];
  }
  return true;
}

function remember(entry) {

}

  const videoConfig=window.tsdVideo.settings(indices);
  const supported=videoConfig.mime;
  if(!supported){targetText('지원되는 녹화 형식이 없습니다');return;}
  const rec={kind:'video',...captureMeta(indices),videoConfig,tracks:[], timer:0, stopLimit:0};

      const track={index,ctx,stream,recorder:null,parts:[]};
      rec.tracks.push(track);
      const recorder=track.recorder=new MediaRecorder(stream,{mimeType:supported,videoBitsPerSecond:spec.bitrate});
      recorder.ondataavailable=e=>{if(e.data.size)track.parts.push(e.data);};
      track.result=new Promise((resolve,reject)=>{
        recorder.onstop=()=>resolve(new Blob(track.parts,{type:recorder.mimeType||'video/webm'}));

      });
      track.result.catch(()=>{});

        track.colorCtx=color.getContext('2d',{willReadFrequently:true});track.colorStream=color.captureStream(videoConfig.fps);

        track.colorRecorder.ondataavailable=e=>{if(e.data.size)track.colorParts.push(e.data);};
        track.colorResult=new Promise((resolve,reject)=>{track.colorRecorder.onstop=()=>resolve(new Blob(track.colorParts,{type:track.colorRecorder.mimeType||'video/webm'}));track.colorRecorder.onerror=e=>reject(e.error||Error('IR 팔레트 녹화 실패'));});track.colorResult.catch(()=>{});
      }
    }
    rec.tracks.forEach(t=>{t.recorder.start(200);t.colorRecorder?.start(200);});
    recording=rec;
    mediaMode='video';
    rec.timer=setInterval(()=>indices.forEach(i=>{if(isVisible(i))maps[i].triggerRepaint();}),Math.round(1000/videoConfig.fps));
    rec.stopLimit=setTimeout(stopRecording,120000);
    updateMediaUI();
    clearCaptureNotice(); // REC indicator remains visible.
  } catch(e) {
    if(recording===rec)recording=null;
    clearInterval(rec.timer);clearTimeout(rec.stopLimit);
    for(const t of rec.tracks) {
      try{if(t.recorder&&t.recorder.state!=='inactive')t.recorder.stop();}catch{}
      try{if(t.colorRecorder&&t.colorRecorder.state!=='inactive')t.colorRecorder.stop();}catch{}
      for(const track of [...t.stream.getTracks(),...(t.colorStream?.getTracks()||[])]){try{track.stop();}catch{}}
    }
    try{updateMediaUI();}catch{}
    targetText('녹화 시작 실패 · '+e.message);
  }
}
async function stopRecording() {
  const rec=recording;
  if(!rec)return;
  recording=null;
  savingRecording=true;
  clearInterval(rec.timer);
  clearTimeout(rec.stopLimit);
  let finishTimer;
  try {
    updateMediaUI();
    for(const track of rec.tracks){if(track.recorder.state!=='inactive')track.recorder.stop();if(track.colorRecorder&&track.colorRecorder.state!=='inactive')track.colorRecorder.stop();}
    const deadline=new Promise((_,reject)=>{finishTimer=setTimeout(()=>reject(Error('녹화 종료 시간 초과')),8000);});
    const [frames,exportFrames]=await Promise.race([Promise.all([Promise.all(rec.tracks.map(t=>t.result)),Promise.all(rec.tracks.map(t=>t.colorResult||t.result))]),deadline]);
    if(frames.some(frame=>frame.size===0)||exportFrames.some(frame=>frame.size===0)||rec.tracks.some(t=>!t.drawn))throw Error('녹화 프레임 없음');
    remember({kind:'video',at:rec.at,duration:Math.max(.1,(Date.now()-rec.at.getTime())/1000),indices:rec.indices,palette:rec.palette,range:rec.range,gain:rec.gain,sensorRange:rec.sensorRange,sourceRange:rec.sourceRange,cameras:rec.cameras,videoConfig:{...rec.videoConfig,tracks:rec.videoConfig.tracks.map(spec=>({...spec,drawnFrames:rec.tracks.find(t=>t.index===spec.index)?.drawn||0}))},frames,exportFrames});
    clearCaptureNotice(); // Recording saved without a center toast.
  } catch(e) {targetText('녹화 저장 실패 · '+e.message);}
  finally {
    clearTimeout(finishTimer);
    savingRecording=false;
    window.tsdVideo?.refresh?.();
    for(const t of rec.tracks){
      try{if(t.recorder&&t.recorder.state!=='inactive')t.recorder.stop();}catch{}
      try{if(t.colorRecorder&&t.colorRecorder.state!=='inactive')t.colorRecorder.stop();}catch{}
      for(const track of [...t.stream.getTracks(),...(t.colorStream?.getTracks()||[])]){try{track.stop();}catch{}}
    }
  }
}

function readingText(values,range) {
  if(!values||!Number.isSafeInteger(values.count)||values.count<=0||!Number.isFinite(values.min)||!Number.isFinite(values.max)||!Number.isFinite(values.sum)||values.min<0||values.max>255||values.min>values.max||values.sum<values.min*values.count-1e-6||values.sum>values.max*values.count+1e-6||!Array.isArray(range)||range.length!==2||!range.every(Number.isFinite)||range[1]<=range[0])return '측정 영역을 다시 선택하세요';
  const temp=n=>(range[0]+n/255*(range[1]-range[0])).toFixed(1)+' °C';
  const edge=n=>n<=0?'≤ '+temp(0)+' (표시 하한)':n>=255?'≥ '+temp(255)+' (표시 상한)':temp(n);
  if(values.count===1)return '모의 온도 '+edge(values.max)+' · 밝기 환산';
  return '모의 최저 '+edge(values.min)+' · 최고 '+edge(values.max)+' · 평균 '+temp(values.sum/values.count)+(values.min===0||values.max===255?' (포화 포함)':'')+' · 밝기 환산';
}

function pixelBounds(a,b,canvas) {
  const x=clamp(Math.floor(Math.min(a.x,b.x)),0,canvas.width-1);
  const y=clamp(Math.floor(Math.min(a.y,b.y)),0,canvas.height-1);
  const w=clamp(Math.ceil(Math.abs(a.x-b.x)),1,canvas.width-x);
  const h=clamp(Math.ceil(Math.abs(a.y-b.y)),1,canvas.height-y);
  return {x,y,w,h};
}

function luminanceStats(ctx,box) {
  const data=ctx.getImageData(box.x,box.y,box.w,box.h).data;
  let min=255,max=0,sum=0,count=0;
  for(let i=0;i<data.length;i+=4) {
    if(!data[i+3])continue;
    const v=(data[i]+data[i+1]+data[i+2])/3;
    min=Math.min(min,v);max=Math.max(max,v);sum+=v;count++;
  }
  return {min,max,sum,count};
}

 const durations=reviewFrames.map(f=>f.video?.duration).filter(d=>Number.isFinite(d)&&d>0);
 return durations.length?Math.min(...durations):Math.max(.1,Number(reviewEntry?.duration)||.1);
}
function tsdReviewUpdateTime(){
 const first=reviewFrames[0]?.video;if(!first)return;
 const duration=tsdReviewDuration();const time=Math.max(0,Math.min(duration,first.currentTime||0));
 $('videoSeek').max=duration;$('videoSeek').value=time;
 $('videoTime').value=reviewTime(time)+' / '+reviewTime(duration);
}
function tsdReviewSeek(time){
 if(!Number.isFinite(time))return;
 const target=Math.max(0,Math.min(tsdReviewDuration(),time));
 for(const f of reviewFrames){const v=f.video;if(v&&v.readyState>=1){try{v.currentTime=target;}catch{}}}
 tsdReviewUpdateTime();
}
function tsdReviewPause(){
 tsdReviewPlayToken++;cancelAnimationFrame(reviewAnimation);reviewAnimation=0;
 for(const f of reviewFrames)f.video?.pause();
 $('videoPlay').textContent='재생';tsdReviewUpdateTime();
}
function playbackTick(now=performance.now()) {
 reviewAnimation=0;
 const first=reviewFrames[0]?.video;
 if(!first || !$('review').open)return;
 const duration=tsdReviewDuration();
 if(first.ended||first.currentTime>=duration){tsdReviewPause();return;}
 if(now-tsdReviewSyncAt>=250){
   tsdReviewSyncAt=now;
   for(const f of reviewFrames.slice(1)){const v=f.video;if(v&&v.readyState>=2&&!v.seeking&&Math.abs(v.currentTime-first.currentTime)>.20){try{v.currentTime=Math.min(duration,first.currentTime);}catch{}}}
 }
 for(const f of reviewFrames){if(!f.video?.seeking)renderReview(f);}
 tsdReviewUpdateTime();
 if(!first.paused)reviewAnimation=requestAnimationFrame(playbackTick);
}
let tsdCountdownTimer=0;
let tsdIntervalTimer=0,tsdIntervalSignature='',tsdIntervalSession=null;
function tsdUpdateIntervalStatus(a=tsdIntervalSession){
 if(!a)return;
 const counts=' · 완료 '+a.done+' · 건너뜀 '+a.skipped+' · 실패 '+a.failed+' · 취소 '+(a.cancelled||0)+(Number.isFinite(a.lastDelay)?' · 최근 지연 '+a.lastDelay+'ms · 최대 '+a.maxDelay+'ms':'');
 if(a.stopped){if(!tsdIntervalSession)$('tsdPhotoStatus').value='종료'+(a.reason?' ('+a.reason+')':'')+counts;return;}
 $('tsdPhotoStatus').value=a.sec+'초'+counts+' · 셔터로 종료';
}
function tsdPhotoSettings(){return {size:$('tsdPhotoSize').value,format:$('tsdPhotoFormat').value,mode:$('tsdPhotoMode').value};}
function tsdStopInterval(reason){
 clearInterval(tsdCountdownTimer);tsdCountdownTimer=0;$('tsdCountdown').hidden=true;
 clearTimeout(tsdIntervalTimer);tsdIntervalTimer=0;
 const a=tsdIntervalSession;tsdIntervalSession=null;tsdIntervalSignature='';
 if(!a)return;
 a.stopped=true;a.reason=reason||'설정 변경';
 // Rev15c: discard only the pending frame owned by the stopped interval session.
 if(photoPending?.intervalSession===a){
   const pending=photoPending;photoPending=null;clearTimeout(pending.timer);
   pending.frames.clear();a.cancelled=(a.cancelled||0)+1;
 }
 tsdUpdateIntervalStatus(a);
}
function tsdPhotoCameraKey(){return ['wide','zoomCamera','irCamera','sbs'].map(id=>$(id).getAttribute('aria-pressed')).join('|');}
function tsdRefreshPhoto(){const wide=$('wide').getAttribute('aria-pressed')==='true',ir=$('irCamera').getAttribute('aria-pressed')==='true';const mode=$('tsdPhotoMode');for(const opt of mode.options){if(opt.value==='lowlight'||opt.value==='panorama'){const disabled=ir||(opt.value==='panorama'&&(!wide||split));if(opt.disabled!==disabled)opt.disabled=disabled;}}if(mode.selectedOptions[0]?.disabled)mode.value='single';const size=$('tsdPhotoSize');const no48=!wide||ir||mode.value==='lowlight'||mode.value==='panorama';if(size.options[2].disabled!==no48)size.options[2].disabled=no48;if(mode.value==='lowlight'&&size.value==='48')size.value='12';if(size.options[1].disabled!==ir)size.options[1].disabled=ir;if(size.selectedOptions[0].disabled)size.value='native';const label=ir?'640 × 512 (IR)':'교육용 기본';if(size.options[0].textContent!==label)size.options[0].textContent=label;if($('tsdPhotoInterval').options[0].disabled!==(size.value==='48'))$('tsdPhotoInterval').options[0].disabled=size.value==='48';if($('tsdPhotoInterval').selectedOptions[0].disabled)$('tsdPhotoInterval').value='3';$('tsdIntervalRow').hidden=$('tsdPhotoMode').value!=='interval';}
function tsdManualShutter(){
 if(mediaMode!=='photo'){tsdStopInterval('모드 변경');return toggleRecording();}
 if(tsdIntervalTimer){tsdStopInterval('사용자 종료');return;}
 if($('tsdPhotoMode').value==='panorama')return tsdPanorama();
 if($('tsdPhotoMode').value!=='interval')return takePhoto();
 if(recording||savingRecording||document.hidden)return;
 const sec=Number($('tsdPhotoInterval').value);
 if(![2,3,5,7,10,15,20,30,60].includes(sec))return;
 const startAt=performance.now(),period=sec*1000;
 const a={sec,done:0,skipped:0,failed:0,cancelled:0,startAt,wallStart:Date.now(),nextAt:startAt+period};
 if(takePhoto(a,startAt)!=='pending')return;
 tsdIntervalSession=a;tsdIntervalSignature=tsdPhotoCameraKey();tsdUpdateIntervalStatus();
 const tick=()=>{const el=$("tsdCountdown");el.hidden=false;el.value=Math.max(0,Math.ceil((a.nextAt-performance.now())/1000))+"초";};tick();tsdCountdownTimer=setInterval(tick,150);
 // Rev15k: phase-locked scheduling; no catch-up bursts when the browser stalls.
 const schedule=()=>{tsdIntervalTimer=setTimeout(run,Math.max(0,a.nextAt-performance.now()));};
 const run=()=>{
  if(tsdIntervalSession!==a||a.stopped)return;
  if(mediaMode!=='photo'||recording||document.hidden||tsdPhotoCameraKey()!==tsdIntervalSignature){tsdStopInterval('촬영 조건 변경');return;}
  const now=performance.now();if(now<a.nextAt){schedule();return;}
  const missed=Math.floor((now-a.nextAt)/period);a.skipped+=missed;
  const due=a.nextAt+missed*period;a.nextAt=due+period;
  const result=takePhoto(a,due);if(result==='busy')a.skipped++;else if(result!=='pending')a.failed++;
  tick();tsdUpdateIntervalStatus();schedule();
 };schedule();
}
function setupMedia() {
  ['tsdPhotoMode','tsdPhotoInterval','tsdPhotoSize','tsdPhotoFormat'].forEach(id=>$(id).addEventListener('change',()=>{tsdCancelPanorama(true);tsdStopInterval();tsdRefreshPhoto();}));
  const photoObs=new MutationObserver(()=>{if(tsdIntervalTimer&&tsdPhotoCameraKey()!==tsdIntervalSignature)tsdStopInterval();tsdRefreshPhoto();});['wide','zoomCamera','irCamera','sbs'].forEach(id=>photoObs.observe($(id),{attributes:true,attributeFilter:['aria-pressed']}));
  document.addEventListener('visibilitychange',()=>{if(document.hidden){tsdCancelPanorama(true);tsdStopInterval();}});tsdRefreshPhoto();
  $('mediaMode').onclick=()=>{
    if(recording)return;
    tsdCancelPanorama(true);
    tsdStopInterval();
    mediaMode=mediaMode==='photo'?'video':'photo';
    updateMediaUI();
  };
  $('shutter').onclick=tsdManualShutter;
  $('playback').onclick=openReview;
  $('reviewClose').onclick=()=>$('review').close();
  $('review').addEventListener('close',()=>{if($('review').open)return;viewRevision++;clearReviewFrames();reviewEntry=null;reviewMeasure=false;$('reviewMeasure').setAttribute('aria-pressed','false');$('reviewReading').value='';});
  $('mediaList').onchange=()=>{const i=$('mediaList').selectedIndex;if(i>=0&&i<recordings.length)loadReview(recordings[i]);};
  $('reviewSensor').onchange=syncReviewControls;
  for(const [id,field] of [['reviewBrightness','brightness'],['reviewContrast','contrast'],['reviewPalette','palette']]) {
    $(id).oninput=()=>{
      const f=activeReviewFrame();
      if(!f)return;
      f[field]=field==='palette'?$(id).value:Number($(id).value);
      renderReview(f);syncReviewControls();
    };
  }
  $('reviewReset').onclick=()=>{
    for(const f of reviewFrames){Object.assign(f,{brightness:100,contrast:100,palette:reviewEntry.palette});renderReview(f);}
    reviewZoom(0);syncReviewControls();
  };
  $('reviewZoomIn').onclick=()=>reviewZoom(.5);
  $('reviewZoomOut').onclick=()=>reviewZoom(-.5);
  $('reviewZoomReset').onclick=()=>reviewZoom(0);
  $('reviewMeasure').onclick=()=>{
    reviewMeasure=!reviewMeasure;
    $('reviewMeasure').setAttribute('aria-pressed',String(reviewMeasure));
    if(!reviewMeasure){reviewFrames.forEach(f=>{f.box.hidden=true;});$('reviewReading').value='';}
  };
  $('reviewExport').onclick=()=>{
    const f=activeReviewFrame();
    if(f)saveFrame(f.canvas,f.index?'IR_adjusted':'EO_adjusted',true);
  };
  $('tsdMediaMeta').onclick=()=>{
    const e=reviewEntry;if(!e)return;
    const info={schema:'TSD-media-v1',kind:e.kind,at:e.at.toISOString(),sensors:e.indices.map((index,i)=>({sensor:index?'IR':'EO',width:e.kind==='photo'?e.frames[i].width:e.videoConfig?.tracks.find(t=>t.index===index)?.width,height:e.kind==='photo'?e.frames[i].height:e.videoConfig?.tracks.find(t=>t.index===index)?.height,mime:e.kind==='video'?e.frames[i].type:null,bytes:e.kind==='video'?e.frames[i].size:null})),photoOutput:e.photoOutput||null,exportTargets:e.kind==='photo'?e.indices.map((index,i)=>{const cfg=e.photoOutput;const dims=tsdPhotoExportDims(cfg,index===1);return {sensor:index?'IR':'EO',width:dims?.[0]??e.frames[i].width,height:dims?.[1]??e.frames[i].height,mime:cfg?.format==='jpeg'?'image/jpeg':'image/png',upscaled:!!dims&&(Math.min(dims[0]/e.frames[i].width,dims[1]/e.frames[i].height)>1),fit:'contain',simulated:true};}):null,exposure:e.exposure||null,intervalTiming:e.intervalTiming||null,videoConfig:e.videoConfig||null,duration:e.duration||null,notice:'TSD simulation; not calibrated sensor data or DJI metadata'};
    download(new Blob([JSON.stringify(info,null,2)],{type:'application/json'}),'TSD_'+e.at.toISOString().replace(/[:.]/g,'-')+'_info.json');
  };
  $('reviewDownload').onclick=()=>{
    const f=activeReviewFrame();
    if(!f)return;
    if(f.video)tsdSaveVideoSensor(f.index);
    else saveFrame(f.source,f.index?'IR_white_source':'EO_source');
  };
  $('videoPlay').onclick=async()=>{
    const first=reviewFrames[0]?.video;if(!first)return;
    if(!first.paused){tsdReviewPause();return;}
    const token=++tsdReviewPlayToken;
    if(first.ended||first.currentTime>=tsdReviewDuration()-.05)tsdReviewSeek(0);
    try{await Promise.all(reviewFrames.map(f=>f.video.play()));}
    catch(e){if(token===tsdReviewPlayToken){tsdReviewPause();$('mediaInfo').textContent+=' · 재생 시작 실패';}return;}
    if(token!==tsdReviewPlayToken||!$('review').open)return;
    $('videoPlay').textContent='일시정지';cancelAnimationFrame(reviewAnimation);playbackTick();
  };
  $('videoSeek').oninput=()=>tsdReviewSeek(Number($('videoSeek').value));
  installLiveMeasure();
  updateMediaUI();
}

// Keep an optional training step distinct from an automatically verified step.
function tutorialStepStatus(isDone,completed,shown,disabled,finalStep) {
  let checkFailed=false;
  if(typeof isDone==='function'&&!completed){
    try {completed=Boolean(isDone());} catch {checkFailed=true;}
  }
  const message=completed?'목표 상태를 확인했습니다. 다음으로 이어가세요.'
    :checkFailed?'자동 확인이 일시적으로 불가능합니다. 직접 확인하거나 건너뛰세요.'
    :!shown?'조작부가 보이지 않으면 먼저 안내에 적힌 메뉴나 비행 상태를 확인하세요.'
    :disabled?'조작부가 잠겨 있습니다. 연결, 시동·이륙 또는 경로 준비 상태를 확인하세요.'
    :typeof isDone==='function'?'목표 상태가 아직 확인되지 않았습니다. 직접 실습하거나 건너뛰세요.':'직접 조작하여 확인하는 단계입니다.';
  const unverified=typeof isDone==='function'&&!completed;
  return {completed,message,label:finalStep?(unverified?'건너뛰고 종료':'실습 종료'):(unverified?'건너뛰기':'다음')};
}

  };

}

let mountainSelection=null;
const normalizedPlaceName=name=>name.normalize('NFC').replace(/\s+/g,'').toLowerCase();
const usableMenuName=name=>/\p{L}{2,}/u.test(name.replace(/\d+\s*(?:m|km|min|h|km²|m²)/gi,''))
  && !/^[\d\s.,:–—\-]*(?:봉|로|점|길|번|구간|코스)(?=$|\s|[\d.,:–—\-])/u.test(name);

function regionalPlaces(){
  const province=activeRegion('province'),district=activeRegion('district'),town=activeRegion('town');
  const districtCodes=new Set(district?[district.key]:[]);
  if(district&&province){
    const prefix=district.name+' ';
    for(const [key,name] of province.children||[])if(name.startsWith(prefix))districtCodes.add(key);
  }
  return placeDb.filter(p=>(!province||p.province===province.key)&&(!district||districtCodes.has(p.district))
    &&(!town||p.town===town.key));
}

function trailBelongsToMountain(trail,mountain){
  const name=mountain.name.trim();
  const parts=name.split(/\s+/);
  const parent=parts.find(word=>/^[가-힣]+(?:산|봉|오름)$/u.test(word));
  const tokens=new Set([name,parent,...(parent&&parts.length===2?[parts[1]]:[])]
    .filter(word=>word&&/^[가-힣\s]{2,}$/u.test(word)&&!['정상','주봉','제1봉','대봉','동봉','서봉','남봉','북봉'].includes(word))
    .map(normalizedPlaceName));
  return [...tokens].some(token=>new RegExp('(^|[^가-힣])'+token+'(?=$|[^가-힣]|등산|둘레|탐방|숲길|코스|정상|구간)').test(trail.name)
    ||normalizedPlaceName(trail.name)===token);
}

function refreshSpots(){
  const region=activeRegion('town')||activeRegion('district')||activeRegion('province');
  const inRegion=region?regionalPlaces():[];
  for(const [category,{id,label}] of Object.entries(placeCategories)){
    const select=$(id),prior=select.value, unique=new Map();
    for(const place of inRegion){
      if(!usableMenuName(place.name))continue;
      const match=category==='trail'?mountainSelection&&place.category==='trail'&&trailBelongsToMountain(place,mountainSelection)
        :category==='recommended'?place.category==='recommended'||place.category==='recreation'||(place.category!=='trail'&&place.rank>=5)
        :place.category===category;
      if(!match)continue;
      const key=place.category+'|'+normalizedPlaceName(place.name)+'|'+(category==='trail'?'':place.district);
      if(!unique.has(key)||unique.get(key).rank<place.rank)unique.set(key,place);
    }
    const items=[...unique.values()].sort((a,b)=>(category==='recommended'?b.rank-a.rank:0)||placeNameOrder.compare(a.name,b.name));
    select.replaceChildren(new Option(label,''));
    select.hidden=select.disabled=!items.length;
    select.removeAttribute('title');
    const fragment=document.createDocumentFragment();
    for(const place of items)fragment.append(new Option(place.name,String(place.index)));
    select.append(fragment);
    if(items.some(place=>String(place.index)===prior))select.value=prior;
    if(category==='mountain'&&mountainSelection&&items.some(place=>place.index===mountainSelection.index)){
      select.value=String(mountainSelection.index);
    }
  }
}

function openDatabasePlace(place,{summit=false}={}){
  if(!place)return;
  placeSelection=place;
  if(place.category==='mountain'){
    mountainSelection=place;
    refreshSpots();
  }
  $('place').value=place.name;
  const region=regionIndex.get(place.district)?.name||regionIndex.get(place.province)?.name;
  const name=[region,place.name].filter(Boolean).join(' ');
  moveToRegion(place.lng,place.lat,summit?name+' 정상':name);
}

let searchRequest = null;
const searchCache = new Map();
function searchRegionNames(input) {
  const name=input.trim();
  for (const id of ['town','district','province']) {
    const a=activeRegion(id);
    if(a?.name===name)return a;
  }
  const matches=[...regionIndex.values()].filter(x=>x.name===name);
  return matches.length===1?matches[0]:null;
}
async function searchPlace(peakOnly){
  const input=$('place').value.trim(),province=activeRegion('province'),district=activeRegion('district');
  searchRequest?.abort();
  searchRequest=null;
  const selectedMountain=$('mountains').value===''?null:placeDb[Number($('mountains').value)];
  if(peakOnly&&selectedMountain&&(!input||selectedMountain.name===input)){
    openDatabasePlace(selectedMountain,{summit:true});return;
  }
  if(peakOnly&&placeSelection?.category==='mountain'&&(!input||placeSelection.name===input)){
    openDatabasePlace(placeSelection,{summit:true});return;
  }
  const admin=input&&!peakOnly?searchRegionNames(input):null;
  if(admin){moveToRegion(admin.lng,admin.lat,admin.name);return;}
  if(!input){setStatusMessage(peakOnly?'산 목록에서 정상을 선택하거나 산 이름을 입력하세요':'찾을 장소명을 입력하세요');return;}
  const norm=normalizedPlaceName;
  const value=norm(input);
  const aliases={한라산:'한라산 백록담 동능',무등산:'무등산 천왕봉',도봉산:'도봉산 자운봉',북한산:'북한산 백운대',
    지리산:'지리산 천왕봉',설악산:'설악산 대청봉',가야산:'가야산 상왕봉',관악산:'관악산 연주대',수락산:'수락산 주봉'};
  const known=aliases[value];
  const inRegion=regionalPlaces();
  const town=activeRegion('town');
  const candidates=inRegion.filter(p=>(!peakOnly||p.category==='mountain')&&norm(p.name).includes(value))
    .sort((a,b)=>Number(b.name===known)-Number(a.name===known)
      ||Number(norm(b.name)===value)-Number(norm(a.name)===value)
      ||(town?Number(b.town===town.key)-Number(a.town===town.key):0)
      ||b.rank-a.rank||placeNameOrder.compare(a.name,b.name));
  if(candidates.length){
    const p=candidates[0],select=$(placeCategories[p.category].id);
    if([...select.options].some(option=>option.value===String(p.index)))select.value=String(p.index);
    openDatabasePlace(p,{summit:peakOnly});
    return;
  }
  searchRequest?.abort();
  const controller=new AbortController();
  searchRequest=controller;
  const query=[input,town?.name,district?.name,province?.name,'대한민국'].filter(Boolean).join(', ');
  setStatusMessage('지명 검색 중 · '+input);
  const timeout=setTimeout(()=>controller.abort('timeout'),8500);
  try {
    let result=searchCache.get(query);
    if(!result) {
      const url=new URL('https://nominatim.openstreetmap.org/search');
      url.search=new URLSearchParams({format:'jsonv2',limit:'12',countrycodes:'kr',q:query});
      const response=await fetch(url,{signal:controller.signal,headers:{Accept:'application/json'}});
      if(!response.ok)throw Error('HTTP '+response.status);
      result=await response.json();
      if(!Array.isArray(result))throw Error('검색 응답 형식 오류');
      searchCache.set(query,result);
    }
    if(searchRequest!==controller)return;
    const point=result.find(p=>validCoordinates(Number(p.lon),Number(p.lat))&&(!peakOnly||p.category==='natural'&&p.type==='peak'));
    if(point)moveToRegion(Number(point.lon),Number(point.lat),point.name||input);
    else setStatusMessage(peakOnly?'산 정상 좌표를 찾지 못했습니다':'검색 결과 없음 · 장소명과 행정구역을 확인하세요');
  } catch(e) {
    if(searchRequest===controller)setStatusMessage('장소 검색 응답 실패 · 행정구역 이동은 사용할 수 있습니다');
  } finally {
    clearTimeout(timeout);
    if(searchRequest===controller)searchRequest=null;
  }
}
