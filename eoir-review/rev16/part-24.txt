   && new Set(saved.channels).size === 7 && Array.isArray(saved.reverse) && saved.reverse.length === 4
        && saved.reverse.every(value => typeof value === "boolean")) {
      return { channels: saved.channels, reverse: saved.reverse };
    }
  } catch { /* Use the default layout if browser storage is unavailable. */ }
  return { channels: defaultChannels, reverse: [false, false, false, false] };
}

function readSession() {
  try {
    if (isNewHostNavigation()) {
      storeSession(null);
      return null;
    }
    const session = JSON.parse(sessionStorage.getItem(hostStorageKey) ?? "null");
    if (session && [session.id, session.hostKey].every(v => typeof v === "string" && token.test(v))
        && Number.isSafeInteger(session.camera?.revision)) return { id: session.id, hostKey: session.hostKey, camera: session.camera };
  } catch { /* The session can still run without storage. */ }
  return null;
}

function storeSession(session) {
  try {
    if (session) sessionStorage.setItem(hostStorageKey, JSON.stringify(session));
    else sessionStorage.removeItem(hostStorageKey);
  } catch { /* The session can still run without storage. */ }
}

async function api(action, data = {}) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), action === "poll" ? 1500 : 4000);
  try {
    const result = await fetch(relayOrigin + "/api/link", {
      method: "POST",
      credentials: "omit",
      cache: "no-store",
      redirect: "manual",
      headers: { "Content-Type": action === "poll" ? "text/plain;charset=UTF-8" : "application/json" },
      body: JSON.stringify({ action, ...data }),
      signal: controller.signal,
    });
    if (requiresSiteSignIn(result)) {
      throw Object.assign(new Error("QR 연결 서비스에 접속할 수 없습니다."), { status: 401 });
    }
    const body = await result.json().catch(() => null);
    if (!result.ok || !body) throw Object.assign(
      new Error(body?.error ?? "QR 연결 서비스와 인터넷 연결을 확인하세요."), { status: result.status });
    return body;
  } finally {
    clearTimeout(timer);
  }
}

  function applyPacket(data, started, freshness = timeoutMs, path = "HTTPS") {
    const packet = readDirectPacket(data?.packet);
    // A different identified sender must not inherit the old video lease.
    if(videoLease && typeof data?.sender==="string" && data.sender && data.sender!==videoLease.sender) videoLease=null;
    if (!packet || typeof data?.sender !== "string" || !data.sender || !Number.isFinite(data.age) || data.age < 0) {
      qrDiagnostics.rejectedInvalid++;
      last = null;
      receivedUntil = 0;
      return;
    }
    const until = started + freshness - data.age;
    if (until <= performance.now()) return;
    if (packet.sequence) {
      if (sequenceSender === data.sender && packet.sequence <= lastSequence) { qrDiagnostics.rejectedSequence++; return; }
      sequenceSender = data.sender;
      lastSequence = packet.sequence;

    const previous = last?.sender === data.sender && (receivedUntil >= started || (packet.sequence && started - receivedUntil < 1000))
      ? last.packet : null;
    if (!eoir.reviewView()) {

          steps.zoom + motion.zoom - baseline.zoom));

        steps.gimbal += packet.gimbal - previous.gimbal;
        steps.zoom += Math.log2(packet.zoom / previous.zoom);
      }
    }
    const reset = packet.commands?.reset ?? 0;
    const resetRequested = reset > (handledReset?.sender === data.sender ? handledReset.count : 0);
    handledReset = { sender: data.sender, count: reset };
    if (suspendedPilot !== data.sender || !packet.arm) suspendedPilot = null;
    last = { sender: data.sender, packet };
    qrDiagnostics.inputPackets++;qrDiagnostics.lastInputAt=performance.now();
    qrDiagnostics.rawAxes=packet.axes.slice();qrDiagnostics.arm=packet.arm;qrDiagnostics.lastInputPath=path;
    receivedUntil = until;
    videoLease = relayVideoLeaseNext(videoLease,session,data.sender,until,performance.now());
    seen = true;
    if (resetRequested) {
      eoir.action("reset");
      return;
    }
    if (previous && packet.capture > previous.capture && eoir.ready()) eoir.photo();
    const modeCount = packet.commands?.flightMode ?? 0;
    const modeRequested = modeCount > (handledMode?.sender === data.sender ? handledMode.count : 0);
    handledMode = { sender: data.sender, count: modeCount };
    if (modeRequested || (previous && previous.mode !== packet.mode)) eoir.mode(packet.mode);
    if (previous && eoir.ready()) {
      for (const [command, count] of Object.entries(packet.commands ?? {})) {
        if (command !== "reset" && command !== "flightMode" && count > (previous.commands?.[command] ?? 0)) eoir.action(command);
      }
    }
  }

  async function poll(current, resume = false) {

      const data = await api("poll", { id: current.id, key: current.hostKey, resume, hostDocument: hostPageId(), direct: supportsDirect });

          .then(() => {

            refreshCamera = false;
            current.camera = updated;
            steps.zoom = 0;
            storeSession(current);
          }).catch(() => {}).finally(() => { if (isCurrent()) configuring = false; });
      }
    } catch (failure) {

        session = null;
        storeSession(null);
        resetLocal();
        dialog.close();
        error = failure.status === 401 ? failure.message : "세션이 끝났습니다. 새 QR을 발급하세요.";
      } else {
        error = failure.message || "조종 연결을 확인 중입니다.";
      }
    }
    updateStatus();
    if (isCurrent()) {
      const nextPoll = directLive() ? 260 : pollMs;
      timer = setTimeout(() => void poll(current), Math.max(0, nextPoll - (performance.now() - started)));
    }
  }

      }

      qrDiagnostics.effectiveAxes=effective;
      return effective;
    },
  };
  window.addEventListener("pagehide", () => {
    if (!session) return;

  if (session) void poll(session, true);
}
